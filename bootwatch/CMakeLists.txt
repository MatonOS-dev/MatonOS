cmake_minimum_required(VERSION 3.22)
project(matonos_bootwatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bootwatch_logic STATIC logic.cpp)
target_include_directories(bootwatch_logic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(matonos-bootwatch main.cpp)
target_link_libraries(matonos-bootwatch PRIVATE bootwatch_logic log)
install(TARGETS matonos-bootwatch RUNTIME DESTINATION bin)
