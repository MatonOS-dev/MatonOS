cmake_minimum_required(VERSION 3.22)
project(matonos_ipc CXX)
file(GLOB_RECURSE MATON_STABLE_AIDL_SOURCES "${MATON_STABLE_AIDL_GENERATED_DIR}/*.cpp")
add_library(matonos-ipc SHARED matonos_ipc.cpp ${MATON_STABLE_AIDL_SOURCES})
target_include_directories(matonos-ipc PUBLIC include ${MATON_STABLE_AIDL_INCLUDE_DIR})
target_compile_features(matonos-ipc PRIVATE cxx_std_20)
target_compile_options(matonos-ipc PRIVATE -Wall -Wextra -Werror -O2 -fvisibility=hidden)
target_link_libraries(matonos-ipc PRIVATE log "${MATON_BINDER_NDK_LIB}")
set_target_properties(matonos-ipc PROPERTIES OUTPUT_NAME matonos-ipc)
install(TARGETS matonos-ipc LIBRARY DESTINATION lib64)

if(MATON_STABLE_AIDL_COUNT LESS 1)
  message(FATAL_ERROR "vendor.matonos.channel stable AIDL generation is required")
endif()
