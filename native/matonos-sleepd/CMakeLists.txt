cmake_minimum_required(VERSION 3.22)
project(matonos_sleepd C)
add_executable(matonos-sleepd matonos-sleepd.c)
target_include_directories(matonos-sleepd PRIVATE "${MATON_IPC_INCLUDE_DIR}")
target_compile_options(matonos-sleepd PRIVATE -Wall -Werror -O2 -fPIE)
target_link_options(matonos-sleepd PRIVATE -pie)
target_link_libraries(matonos-sleepd PRIVATE log "${MATON_IPC_LIBRARY}")
if(MATON_AIDL_COUNT GREATER 0)
  file(GLOB_RECURSE MATON_AIDL_CPP "${MATON_AIDL_GENERATED_DIR}/*.cpp")
  target_sources(matonos-sleepd PRIVATE ${MATON_AIDL_CPP})
  target_include_directories(matonos-sleepd PRIVATE "${MATON_AIDL_GENERATED_DIR}" "${MATON_AIDL_INCLUDE_DIR}")
  target_link_libraries(matonos-sleepd PRIVATE binder_ndk)
endif()
install(TARGETS matonos-sleepd RUNTIME DESTINATION .)
