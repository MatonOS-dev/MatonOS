cmake_minimum_required(VERSION 3.22)
project(matonos_wifid CXX)

add_executable(matonos-wifid matonos-wifid.cpp)
target_include_directories(matonos-wifid PRIVATE "${MATON_IPC_INCLUDE_DIR}")
target_compile_features(matonos-wifid PRIVATE cxx_std_17)
target_compile_options(matonos-wifid PRIVATE -Wall -Wextra -Werror -O2 -fPIE)
target_link_options(matonos-wifid PRIVATE -pie)
target_link_libraries(matonos-wifid PRIVATE log "${MATON_IPC_LIBRARY}")
if(MATON_AIDL_COUNT GREATER 0)
  file(GLOB_RECURSE MATON_AIDL_CPP "${MATON_AIDL_GENERATED_DIR}/*.cpp")
  target_sources(matonos-wifid PRIVATE ${MATON_AIDL_CPP})
  target_include_directories(matonos-wifid PRIVATE "${MATON_AIDL_GENERATED_DIR}" "${MATON_AIDL_INCLUDE_DIR}")
  target_link_libraries(matonos-wifid PRIVATE binder_ndk)
endif()
install(TARGETS matonos-wifid RUNTIME DESTINATION .)
