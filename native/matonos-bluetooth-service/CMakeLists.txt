cmake_minimum_required(VERSION 3.22)
project(matonos_bluetooth_service CXX)
if(MATON_STABLE_AIDL_COUNT LESS 1)
  message(FATAL_ERROR "Bluetooth service requires generated VINTF NDK AIDL sources")
endif()
file(GLOB_RECURSE AIDL_CPP "${MATON_STABLE_AIDL_GENERATED_DIR}/*.cpp")
add_executable(matonos-bluetooth-service main.cpp BluetoothHci.cpp ${AIDL_CPP})
target_include_directories(matonos-bluetooth-service PRIVATE
  "${MATON_STABLE_AIDL_GENERATED_DIR}" "${MATON_STABLE_AIDL_INCLUDE_DIR}")
target_compile_features(matonos-bluetooth-service PRIVATE cxx_std_17)
target_compile_options(matonos-bluetooth-service PRIVATE -Wall -Wextra -Werror -O2 -fPIE)
target_link_options(matonos-bluetooth-service PRIVATE -pie)
target_link_libraries(matonos-bluetooth-service PRIVATE "${MATON_BINDER_NDK_LIB}" log)
install(TARGETS matonos-bluetooth-service RUNTIME DESTINATION .)
